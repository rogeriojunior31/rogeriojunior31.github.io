{{- /* Metadados do projeto acompanham a mesma versão dos docs importados. */ -}} {{- $projects := slice -}} {{- range hugo.Data.docs -}} {{- $project := . -}} {{- if .url -}} {{- $urls := merge (dict "fallback" .url) (.urls | default dict) -}} {{- range $language, $url := $urls -}} {{- $parsed := urls.Parse $url -}} {{- if not (and (eq $parsed.Scheme "https") $parsed.Host) -}}{{ errorf "Docs %s: url deve ser uma URL HTTPS absoluta (%s)" $project.slug $language }}{{- end -}} {{- end -}} {{- $localized := index $urls site.Language.Lang -}} {{- $language := .urlLanguage | default "unknown" -}} {{- if $localized }}{{ $language = site.Language.Lang }}{{ end -}} {{- $project = merge $project (dict "url" ($localized | default .url) "urlLanguage" $language) -}} {{- end -}} {{- if not .url -}} {{- with resources.Get (printf "docs/%s/site.json" .slug) -}} {{- $meta := .Content | transform.Unmarshal -}} {{- if ne (int ($meta.schema | default 0)) 1 -}} {{- errorf "Docs %s: site.json exige schema 1" $project.slug -}} {{- end -}} {{- if not (strings.TrimSpace ($meta.name | default "")) -}}{{ errorf "Docs %s: site.json exige name" $project.slug }}{{- end -}} {{- range hugo.Sites -}} {{- if not (strings.TrimSpace (index ($meta.summary | default dict) .Language.Lang | default "")) -}} {{- errorf "Docs %s: site.json exige summary.%s" $project.slug .Language.Lang -}} {{- end -}} {{- end -}} {{- $required := $meta.requiredTranslations | default slice -}} {{- $languages := slice -}}{{- range hugo.Sites }}{{ $languages = $languages | append .Language.Lang }}{{ end -}} {{- range $required -}} {{- if or (eq . "en") (not (in $languages .)) -}}{{ errorf "Docs %s: requiredTranslations contém idioma inválido %s" $project.slug . }}{{- end -}} {{- end -}} {{- $project = merge $project (dict "name" $meta.name "summary" $meta.summary "requiredTranslations" $required) -}} {{- end -}} {{- end -}} {{- $projects = $projects | append $project -}} {{- end -}} {{- return $projects -}}
